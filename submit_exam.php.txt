<?php
session_start();
require_once 'db.php';

// Security check: Ensure student is logged in
if (!isset($_SESSION['student_id'])) {
    header("Location: login.php");
    exit();
}

$student_id = $_SESSION['student_id'];
$student_name = $_SESSION['student_name'];
$reg_number = $_SESSION['reg_number'];

$score = 0;
$total_questions = 0;
$percentage = 0;
$status = "FAILED";

if ($_SERVER['REQUEST_METHOD'] == 'POST') {
    try {
        // Fetch all questions from the database
        $stmt = $pdo->query("SELECT * FROM questions");
        $questions = $stmt->fetchAll();
        $total_questions = count($questions);

        foreach ($questions as $q) {
            $q_id = $q['id'];
            
            // Check whichever column name exists in your database table
            $correct_ans = "";
            if (isset($q['correct_answer'])) {
                $correct_ans = trim(strtoupper($q['correct_answer']));
            } elseif (isset($q['correct_option'])) {
                $correct_ans = trim(strtoupper($q['correct_option']));
            }

            // Check if the student answered this question correctly
            if (isset($_POST["question_" . $q_id])) {
                $student_ans = trim(strtoupper($_POST["question_" . $q_id]));
                
                if ($student_ans === $correct_ans) {
                    $score++;
                }
            }
        }

        // Calculate percentage securely
        if ($total_questions > 0) {
            $percentage = ($score / $total_questions) * 100;
        }
        
        // Pass/Fail evaluation threshold (50%)
        if ($percentage >= 50) {
            $status = "PASSED";
        } else {
            $status = "FAILED";
        }

        // Attempt to save result into database safely (won't crash the page if table is missing)
        try {
            $stmt_res = $pdo->prepare("INSERT INTO results (student_id, score, total, status) VALUES (?, ?, ?, ?)");
            $stmt_res->execute([$student_id, $score, $total_questions, $status]);
        } catch (Exception $ex) {
            // Silently pass if results table isn't created yet so calculation still shows
        }

    } catch (PDOException $e) {
        $score = 0;
        $total_questions = 0;
        $percentage = 0;
        $status = "FAILED";
    }
} else {
    header("Location: dashboard.php");
    exit();
}
